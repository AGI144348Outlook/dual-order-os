class IndexCard:
    def __init__(self, card_id: str, lineage_hash: str):
        # --- Invariant Bounding Box (The LCD) ---
        self.card_id = card_id
        self.lineage_hash = lineage_hash  # Tied to the 'ר' (Resh) root
        
        # --- Short-Edge I Ching Ports (6-bit binary masks: 0 to 63) ---
        # Each port maps to a short edge acting as a physical router
        self.ports = {
            "ingress": 0,      # Edge 1: What inputs this card accepts
            "egress": 0,       # Edge 2: Where output routes next
            "domain": 0,       # Edge 3: Topological native math (Symbolic, Spatial, etc.)
            "jurisdiction": 0  # Edge 4: Access level / registry permissions
        }
        
        # --- Hebrew Operational Root Metadata (ע - ת Stamps) ---
        self.metadata = {
            "ayin_perceive": None,       # Observational baseline
            "pe_name": None,             # Agent descriptor
            "resh_identify": lineage_hash, # Genesis hash
            "tzadi_index": [],           # Pointer addresses / paperclips / growth slots
            "qof_concordance": [],       # Footnotes / cross-references
            "shin_transformation": None, # Algorithmic topology type
            "tav_stamp": "DORMANT"       # Jurisdictional validation status
        }
        
        self.payload = None              # The actual algorithmic logic or data
        self.is_active = False           # Tactile state: Flipped/Tilted flag

    def stamp_edge_port(self, port_name: str, hexagram_value: int):
        """Punches or sets the 6-bit I Ching binary pattern along a short edge."""
        if port_name in self.ports and 0 <= hexagram_value <= 63:
            self.ports[port_name] = hexagram_value

    def apply_root_stamp(self, root: str, value):
        """Applies an operational stamp (ע-ת) to modify state without destroying lineage."""
        root_map = {
            'ע': 'ayin_perceive',
            'פ': 'pe_name',
            'ר': 'resh_identify',
            'צ': 'tzadi_index',
            'ק': 'qof_concordance',
            'ש': 'shin_transformation',
            'ת': 'tav_stamp'
        }
        if root in root_map:
            key = root_map[root]
            if isinstance(self.metadata[key], list):
                self.metadata[key].append(value)
            else:
                self.metadata[key] = value

class Deck:
    def __init__(self, name: str):
        self.name = name
        self.cards = []

    def append_card(self, card: IndexCard):
        """Monotonic growth: cards are added to extend scope, never overwritten."""
        self.cards.append(card)

class EdgeNotchPin:
    def __init__(self, target_port: str, target_hexagram: int):
        self.target_port = target_port
        self.target_hexagram = target_hexagram

    def sort_deck(self, deck: Deck) -> list:
        """Simulates dropping the sorting needle through a deck of index cards."""
        dropped_cards = []
        retained_cards = []
        
        for card in deck.cards:
            # If the card's short-edge port matches the pin's hexagram, it drops free
            if card.ports.get(self.target_port) == self.target_hexagram:
                dropped_cards.append(card)
            else:
                retained_cards.append(card)
                
        print(f"[PIN DROP] Filtered port '{self.target_port}' for hexagram {self.target_hexagram}. "
              f"Dropped {len(dropped_cards)} matching cards.")
        return dropped_cards
